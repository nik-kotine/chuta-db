-- 2.1.1  Heap vs. secuencial paginado
CREATE TABLE ventas_heap (id INT PRIMARY KEY, cliente VARCHAR(50), monto FLOAT, pagado BOOL) USING HEAP;
CREATE TABLE ventas_seq (id INT PRIMARY KEY, monto FLOAT) USING SEQUENTIAL;

-- 2.1.2  Indices
CREATE INDEX ON ventas_seq (id) USING BTREE CLUSTERED;
CREATE INDEX ON ventas_heap (cliente) USING BTREE;
CREATE INDEX ON ventas_heap (id) USING HASH;

-- 2.1.3  Consultas
SELECT * FROM ventas_heap WHERE id = 4021;
SELECT id, cliente FROM ventas_seq WHERE id BETWEEN 1000 AND 2000;
SELECT * FROM ventas_heap WHERE monto > 500.5;
SELECT * FROM ventas_heap WHERE monto > 100 AND pagado = TRUE OR cliente = 'ACME';
SELECT * FROM ventas_heap WHERE (monto > 100 OR monto < 10) AND pagado = FALSE;
SELECT id FROM ventas_seq ORDER BY id ASC LIMIT 20;
SELECT * FROM ventas_heap ORDER BY monto DESC;
SELECT cliente, COUNT(*), SUM(monto), AVG(monto), MIN(monto), MAX(monto) FROM ventas_heap GROUP BY cliente;
SELECT ventas_heap.id, ventas_seq.monto FROM ventas_heap JOIN ventas_seq ON ventas_heap.id = ventas_seq.id WHERE ventas_heap.monto >= 50;
INSERT INTO ventas_heap VALUES (4021, 'Juan O''Brien', 1250.75, TRUE);
DELETE FROM ventas_heap WHERE id = 4021;

-- 2.1.4  Transacciones
BEGIN TRANSACTION;
INSERT INTO ventas_seq VALUES (999, 10.5);
DELETE FROM ventas_seq WHERE id = 999;
END TRANSACTION
